function generateGame(){ let title=document.getElementById("gameTitle").value; let time=document.getElementById("gameTime").value; let selected=[]; document.querySelectorAll(".selected").forEach(x=>{ selected.push(x.innerText); }); localStorage.setItem("gameTitle",title); localStorage.setItem("gameTime",time); localStorage.setItem("gameNames",JSON.stringify(selected)); alert("Game Generated Successfully"); } function reGenerateGame(){ let time=prompt("Enter New Date & Time"); if(time){ localStorage.setItem("gameTime",time); alert("New Time Saved"); } } function cancelGame(){ if(confirm("Cancel this game?")){ localStorage.removeItem("gameTitle"); localStorage.removeItem("gameTime"); alert("Game Cancelled"); } }
